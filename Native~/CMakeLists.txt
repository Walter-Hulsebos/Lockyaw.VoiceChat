cmake_minimum_required(VERSION 3.21)
project(LockyawVoiceCapture LANGUAGES CXX)
add_library(LockyawVoiceCapture SHARED VoiceCapture.cpp)
target_compile_features(LockyawVoiceCapture PRIVATE cxx_std_17)
set_target_properties(LockyawVoiceCapture PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    set_property(TARGET LockyawVoiceCapture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(APPLE)
    target_link_libraries(LockyawVoiceCapture PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
elseif(UNIX)
    target_link_libraries(LockyawVoiceCapture PRIVATE dl pthread m)
endif()
